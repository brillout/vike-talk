// ChatRoom.telefunc.ts
// Environment: server // [!code highlight]

import { createChannel } from 'telefunc'

export async function onChatInit(chatRoomId: string) {
  const channel = createChannel<string, string>()

  channel.listen((message) => { // [!code highlight:#c0ebb8]
    Database.ChatRoom.insert({ chatRoomId, message })
  })

  Database.ChatRoom.listen({ chatRoomId}, (message) => {
    channel.send(message) // [!code highlight:#b8d4eb]
  })

  return { channel: channel.client }
}
