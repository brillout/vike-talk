```ts
import ReactDOMServer from 'react-dom/server'

export function onRenderHtml(pageContext) {
  const { Page } = pageContext
  const pageHtml = ReactDOMServer.renderToString(<Page/>)
  return `
    <html>
      <body>
        <div id="root">${pageHtml}</div>
      </body>
    </html>`
}
```
