      export default function Article() {
        const fetcher = useFetcher()

        const handleDelete = () => {
          fetcher.submit(
            { article_id },
            { method: "POST", action: "/article/delete" }
          )
        }

        return <button onClick={handleDelete}>Delete</button>
      }
